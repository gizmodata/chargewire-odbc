cmake_minimum_required(VERSION 3.28)
cmake_policy(SET CMP0169 OLD)
project(chargewire_odbc VERSION 1.0.0 LANGUAGES C CXX)
set(CHARGEWIRE_ODBC_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR})
if(NOT DEFINED CHARGEWIRE_ODBC_LIBRARY_NAME)
  set(CHARGEWIRE_ODBC_LIBRARY_NAME chargewire-odbc)
endif()
if(NOT DEFINED CHARGEWIRE_ODBC_PRODUCT_NAME)
  set(CHARGEWIRE_ODBC_PRODUCT_NAME "Chargewire ODBC Driver")
endif()
if(NOT DEFINED CHARGEWIRE_ODBC_PROFILE_SOURCE)
  set(CHARGEWIRE_ODBC_PROFILE_SOURCE ${CHARGEWIRE_ODBC_SOURCE_DIR}/chargewire/default_profile.cc)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  # fmt and the Unicode regression fixtures require a UTF-8 execution charset.
  # Boost.Beast's TLS transport exceeds the default COFF section limit.
  add_compile_options(/utf-8 "$<$<COMPILE_LANGUAGE:CXX>:/bigobj>")
endif()
if(NOT DEFINED CHARGEWIRE_ODBC_PRODUCT_VERSION)
  set(CHARGEWIRE_ODBC_PRODUCT_VERSION ${PROJECT_VERSION})
endif()
string(REPLACE "." ";" CHARGEWIRE_ODBC_VERSION_PARTS "${CHARGEWIRE_ODBC_PRODUCT_VERSION}")
list(GET CHARGEWIRE_ODBC_VERSION_PARTS 0 CHARGEWIRE_ODBC_VERSION_MAJOR)
list(GET CHARGEWIRE_ODBC_VERSION_PARTS 1 CHARGEWIRE_ODBC_VERSION_MINOR)
list(GET CHARGEWIRE_ODBC_VERSION_PARTS 2 CHARGEWIRE_ODBC_VERSION_PATCH)
set(CHARGEWIRE_ODBC_VERSION_TWEAK 0)
add_compile_definitions(CHARGEWIRE_ODBC_VERSION="${CHARGEWIRE_ODBC_PRODUCT_VERSION}" CHARGEWIRE_VERSION="1.0.1" FMT_USE_CONSTEVAL=0)
find_package(Threads REQUIRED)
set(OPENSSL_USE_STATIC_LIBS ON)
find_package(OpenSSL 3 REQUIRED)
set(Boost_USE_STATIC_LIBS ON)
find_package(Boost 1.85 CONFIG REQUIRED COMPONENTS json url locale)
find_package(ODBC REQUIRED)
find_path(RapidJSON_INCLUDE_DIRS rapidjson/document.h REQUIRED)
include_directories(${Boost_INCLUDE_DIRS} ${ODBC_INCLUDE_DIRS} ${RapidJSON_INCLUDE_DIRS})
# The driver and pyodbc both use unixODBC, including on macOS (16-bit SQLWCHAR).
include(FetchContent)
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)
find_package(Arrow CONFIG REQUIRED)
find_package(ArrowCompute CONFIG REQUIRED)
if(WIN32)
  add_compile_definitions(ARROW_STATIC ARROW_COMPUTE_STATIC NOMINMAX)
endif()
enable_testing()
add_library(chargewire_client STATIC third_party/chargewire/cpp/src/client/client.cc third_party/chargewire/cpp/src/client/legacy_dialect.cc third_party/chargewire/cpp/src/client/transport.cc)
target_include_directories(chargewire_client PUBLIC third_party/chargewire/cpp/include PRIVATE third_party/chargewire/cpp/src/client)
target_link_libraries(chargewire_client PUBLIC Boost::json Boost::url Arrow::arrow_static OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
add_subdirectory(odbcabstraction)
# The abstraction owns DSN/profile calls. Propagate its installer dependency so
# GNU ld sees it after the static archive, including in unit-test executables.
if(UNIX)
  find_library(ODBCINST_LIBRARY odbcinst REQUIRED)
  target_link_libraries(odbcabstraction PUBLIC ${ODBCINST_LIBRARY})
elseif(WIN32)
  target_link_libraries(odbcabstraction PUBLIC odbccp32 legacy_stdio_definitions)
endif()
add_subdirectory(chargewire)
add_subdirectory(odbc_entry)
foreach(suite metadata transactions)
  add_executable(test_${suite} integration_tests/test_${suite}.c)
  target_link_libraries(test_${suite} PRIVATE ODBC::ODBC)
endforeach()
install(TARGETS chargewire-odbc LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES LICENSE NOTICE DESTINATION share/${CHARGEWIRE_ODBC_LIBRARY_NAME})
