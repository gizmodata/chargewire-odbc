file(GLOB SPI_SOURCES CONFIGURE_DEPENDS "*.cc" "accessors/*.cc")
list(FILTER SPI_SOURCES EXCLUDE REGEX "_test\\.cc$")
if(WIN32)
  file(GLOB WINDOWS_SOURCES "config/*.cc" "ui/*.cc")
  list(APPEND SPI_SOURCES ${WINDOWS_SOURCES})
else()
  list(FILTER SPI_SOURCES EXCLUDE REGEX "system_dsn\\.cc$")
endif()
add_library(hardcharger_odbc_spi_impl STATIC ${SPI_SOURCES})
target_include_directories(hardcharger_odbc_spi_impl PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_SOURCE_DIR}/odbcabstraction/include)
target_link_libraries(hardcharger_odbc_spi_impl PUBLIC odbcabstraction warpsql_client ArrowCompute::arrow_compute_static Boost::locale)
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS "*_test.cc" "accessors/*_test.cc")
add_executable(hardcharger_odbc_spi_impl_test ${TEST_SOURCES})
target_link_libraries(hardcharger_odbc_spi_impl_test PRIVATE hardcharger_odbc_spi_impl gtest_main)
add_test(NAME odbc-unit COMMAND hardcharger_odbc_spi_impl_test)
