file(GLOB SPI_SOURCES CONFIGURE_DEPENDS "*.cc" "accessors/*.cc")
list(FILTER SPI_SOURCES EXCLUDE REGEX "_test\\.cc$")
list(FILTER SPI_SOURCES EXCLUDE REGEX "default_profile\\.cc$")
list(APPEND SPI_SOURCES ${WARP_ODBC_PROFILE_SOURCE})
file(GLOB CONFIG_SOURCES "config/*.cc")
list(APPEND SPI_SOURCES ${CONFIG_SOURCES})
if(WIN32)
  file(GLOB WINDOWS_SOURCES "ui/*.cc")
  list(APPEND SPI_SOURCES ${WINDOWS_SOURCES})
else()
  list(FILTER SPI_SOURCES EXCLUDE REGEX "system_dsn\\.cc$")
endif()
add_library(warpsql_odbc_spi_impl STATIC ${SPI_SOURCES})
target_include_directories(warpsql_odbc_spi_impl PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/include/warpsql ${WARP_ODBC_SOURCE_DIR}/odbcabstraction/include)
target_link_libraries(warpsql_odbc_spi_impl PUBLIC odbcabstraction warpsql_client ArrowCompute::arrow_compute_static Boost::locale)
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS "*_test.cc" "accessors/*_test.cc")
add_executable(warpsql_odbc_spi_impl_test ${TEST_SOURCES} ${WARP_ODBC_PROFILE_TEST_SOURCES})
target_link_libraries(warpsql_odbc_spi_impl_test PRIVATE warpsql_odbc_spi_impl gtest_main)
add_test(NAME odbc-unit COMMAND warpsql_odbc_spi_impl_test)
